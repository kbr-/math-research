\section{Proof of the main theorem}\label{sec:main}

\begin{theorem}[The formula below \(2^{n-1}\)]\label{thm:main-formal}
Let \(k\ge1\).
\begin{enumerate}
 \item For every \(n\ge\lfloor\log_2k\rfloor+2\), every \(k\)-admissible
 \(P\in\Ft[x_1,\ldots,x_n]\) has \(\deg P\ge2k+n-2-\lfloor\log_2k\rfloor\).
 \item For every \(n\ge1\), some \(k\)-admissible \(P\in\Ft[x_1,\ldots,x_n]\) has
 \(\deg P\le2k+n-2-\lfloor\log_2k\rfloor\).
\end{enumerate}
Hence \(D(n,k)=2k+n-2-\lfloor\log_2k\rfloor\) whenever \(n\ge\lfloor\log_2k\rfloor+2\),
equivalently \(k<2^{n-1}\), and a polynomial of exactly this degree exists
\lean{claims/BinaryMultiplicityDegreeFormula.lean}.
\end{theorem}

\begin{proof}
Part~1 is Theorem~\ref{thm:lower}.

For part~2, let \(J=\lfloor\log_2k\rfloor\), so \(2^J\le k\), and let \(\ell=k-2^J<k\).
The polynomial of Theorem~\ref{thm:per-order} is \(k\)-admissible with origin order
\(\ell\), and its degree is at most
\[
 n+2k-2-s_2(2^J-1)=n+2k-2-J
\]
by Lemma~\ref{lem:digits}(4).

For \(k\ge1\), \(k<2^{n-1}\) is equivalent to \(\lfloor\log_2k\rfloor<n-1\), which is
\(n\ge\lfloor\log_2k\rfloor+2\).
\end{proof}

\begin{corollary}\label{cor:extended}
For \(n\ge1\) and \(1\le k<3\cdot2^{n-1}\), \(D(n,k)=2k+n-2-\lfloor\log_2k\rfloor\)
\lean{claims/BinaryMultiplicityDegreeExtendedRange.lean}.
\end{corollary}

\begin{proof}
For \(k<2^{n-1}\) this is Theorem~\ref{thm:main-formal}. The upper bound holds in every
dimension, so it remains to match it with Proposition~\ref{prop:cover-bound} in two cases.
\begin{itemize}
 \item If \(2^{n-1}\le k<2^n\), then \(\lfloor\log_2k\rfloor=n-1\) and
 \(\lfloor k/2^{n-1}\rfloor=1\), so both bounds equal \(2k-1\).
 \item If \(2^n\le k<3\cdot2^{n-1}\), then \(\lfloor\log_2k\rfloor=n\) and
 \(\lfloor k/2^{n-1}\rfloor=2\), so both equal \(2k-2\). \qedhere
\end{itemize}
\end{proof}

At \(k=3\cdot2^{n-1}\) the bounds are \(2k-3\) and \(2k-2\), and the gap grows with
\(k\). Proposition~\ref{prop:cover-bound} is therefore attained exactly for
\(2^{n-2}\le k<3\cdot2^{n-1}\).

\begin{remark}[Which origin orders are extremal]
Let \(n\ge\lfloor\log_2k\rfloor+2\), or more generally \(k<3\cdot2^{n-1}\), so that
\(D(n,k)\) is known. Theorem~\ref{thm:per-order} gives degree
\(n+2k-2-s_2(k-\ell-1)\) at origin order \(\ell\). This equals \(D(n,k)\) exactly
when \(s_2(k-\ell-1)=\lfloor\log_2k\rfloor\), for instance \(\ell=k-2^{\lfloor\log_2k\rfloor}\).
The 69 per-order minima computed exactly for \(n\le7\) in the
\href{https://kbr.is-a.dev/math-research/branches/binary-multiplicity-degree/#binary-multiplicity-per-order-formula}{research notebook}
all equal \(n+2k-2-s_2(k-\ell-1)\), including 15 values with \(n<k-1\). That
computation is evidence, not part of the formalized result.
\end{remark}

\subsection*{What does not follow}

\begin{itemize}
 \item \textbf{Per-order minima below \(n=k-1\).} For \(n<k-1\) we do not prove
 that \(n+2k-2-s_2(k-\ell-1)\) is the least degree at an origin order \(\ell\)
 with \(s_2(k-\ell-1)<\lfloor\log_2k\rfloor\). For \(n\ge k-1\) this is part of
 Menezes' theorem~\cite{Men26}.
 \item \textbf{Large \(k\).} For \(k\ge3\cdot2^{n-1}\) the lower bound of
 Proposition~\ref{prop:cover-bound} and the upper bound of Theorem~\ref{thm:main-formal}(2)
 differ, and we do not determine \(D(n,k)\).
 \item \textbf{Other fields and grids.} The lower bound uses the binary setting in
 several places: the cancellation of the \(x_0\)-free part in \(Q(0,x')+Q(1,x')\),
 the finite number of linear forms in Lemma~\ref{lem:linear-form}, and \(S=\Ft\) in
 Proposition~\ref{prop:cover-bound}. The upper bound uses characteristic two in the
 telescoping identity. We make no claim about other fields or grids.
\end{itemize}
