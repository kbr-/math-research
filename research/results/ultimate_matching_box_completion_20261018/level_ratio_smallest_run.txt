import math,json,numpy as np
import box_completion as b
out=[]
for N in (4096,16384):
  for seed in range(3):
    rng=np.random.default_rng(7000+N+seed); X=rng.uniform(0,1,(N,2)); g=int(math.isqrt(N))
    cells=np.stack(np.meshgrid(np.arange(g),np.arange(g),indexing='ij'),-1).reshape(-1,2); Y=(cells+rng.uniform(0,1,(N,2)))/g
    s1=math.sqrt(math.log(N)/N); A1=b.graph(X,Y,0.5*s1); m1=N-int((b.maximum_bipartite_matching(A1)>=0).sum())
    for c in (2,4,8,16):
      Ac=b.graph(X,Y,0.5*c**0.25*s1); mc=N-int((b.maximum_bipartite_matching(Ac)>=0).sum())
      _,det=b.certificate(A1,Ac); r=dict(N=N,seed=seed,c=c,m1=m1,mc=mc,cert=det['cert'],nuD=det['nuD'],BX=det['BX'],nuBX=det['nuBX'],BY=det['BY'],nuBY=det['nuBY'])
      out.append(r); print(r,flush=True)
json.dump(out,open('research/results/ultimate_matching_box_completion_20261018/level_ratio_smallest.json','w'),indent=1)
