import numpy as np
print('Smallest nontrivial case: children with two positions each, values R+d, R-d and -R+d, -R-d; uniform shift m on each child; leg 2m on one child area (B = 0).')
print('cost per two positions: (R+d-m)^4+(R-d-m)^4 for each child (by symmetry twice) + 2*(2m)^4; linear rule R/3')
ms=np.linspace(-1,3,400001)
for R in (0.25,1.0,2.0):
  for d in (0.0,1.0,2.0):
    f=2*((R+d-ms)**4+(R-d-ms)**4)+2*(2*ms)**4
    print('R=%.2f d=%.1f argmin m=%.4f  R/3=%.4f  ratio m/(R/3)=%.3f'%(R,d,ms[f.argmin()],R/3,ms[f.argmin()]/(R/3)))
